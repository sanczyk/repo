            // Se comprueba si la predicción es incorrecta:
            // - Si label=1 (Rojo) y sum < 0 (predice Azul) -> (1 * sum) es negativo -> ERROR
            // - Si label=-1 (Azul) y sum > 0 (predice Rojo) -> (-1 * sum) es negativo -> ERROR
            // - Si los signos coinciden, (label * sum) es positivo -> ACIERTO
            
            let misclassified = (sum * data.label <= 0);
            
            if (misclassified) {
                epochAdjustments++;
                
                // 4. AJUSTE DE PESOS Y SESGO
                // El ajuste es (label * input * tasa_aprendizaje)
                // (label es 1 o -1, lo que "empuja" los pesos en la dirección correcta)
                
                for (let j = 0; j < weights.length; j++) {
                    // El "delta" o cambio es data.label (1 o -1)
                    const deltaWeight = data.label * inputs[j] * learningRate;
                    showFloatingText(document.getElementById(`weight-bar-${['R','G','B'][j]}-container`), deltaWeight);
                    weights[j] += deltaWeight;
                }
                // El sesgo también se ajusta
                const deltaBias = data.label * learningRate;
                showFloatingText(document.getElementById('bias-container'), deltaBias);
                bias += deltaBias;

Función de Predicción (predict()) con Confianza


// Valor de "suma" que consideramos 100% de confianza
const maxConfidenceSum = 2.5;

// Calcula un porcentaje de confianza basado en la magnitud de la suma
function calculateConfidence(sum) {
    // abs(sum) es la "distancia a 0". Más grande = más confianza.
    const confidence = (Math.abs(sum) / maxConfidenceSum) * 100;
    // No puede ser más de 100%
    return Math.min(100, confidence).toFixed(0);
}

// Función para predecir un nuevo color (se llama en tiempo real)
function predict() {
    // ... (obtener r, g, b del colorPicker) ...
    const inputs = [r / 255, g / 255, b / 255];

    // 1. CÁLCULO DE LA SUMA PONDERADA
    let sum = bias;
    for (let j = 0; j < inputs.length; j++) {
        sum += inputs[j] * weights[j];
    }
    
    // 2. FUNCIÓN DE ACTIVACIÓN
    const prediction = sum >= 0 ? 1 : -1;
    
    // 3. CÁLCULO DE CONFIANZA
    const confidence = calculateConfidence(sum);

    // 4. MOSTRAR RESULTADO
    const resultDiv = document.getElementById('prediction-result');
    resultDiv.style.backgroundColor = hexColor;

    if (prediction === 1) {
        resultDiv.innerHTML = `Predicción: ROJO (${confidence}%)`;
        resultDiv.style.color = isColorDark(r, g, b) ? 'white' : 'black';
    } else {
        resultDiv.innerHTML = `Predicción: AZUL (${confidence}%)`;
        resultDiv.style.color = isColorDark(r, g, b) ? 'white' : 'black';
    }
}
